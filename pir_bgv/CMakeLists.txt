set(USE_HARDWARE OFF)
if(USE_HARDWARE)
    add_definitions(-DPIR_USE_HARDWARE)
    message(STATUS "pir hardware version")
else()
    message(STATUS "pir software version")
endif()

file(GLOB PIR_BGV_SOURCE_FILES "*.cpp" "*.c")
add_executable(pir_bgv ${PIR_BGV_SOURCE_FILES})
target_link_libraries(pir_bgv PUBLIC poseidon_shared)
message(STATUS "PIR_BGV build success")

option(PIR_BGV_BUILD_TESTS "Build tests for PIR of BGV" OFF)
if(PIR_BGV_BUILD_TESTS)
    add_executable(pir_bgv_tests)
    add_subdirectory(tests)
    target_include_directories(pir_bgv_tests PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}>)
    target_link_libraries(pir_bgv_tests poseidon_shared GTest::gtest)
endif()
message(STATUS "PIR_BGV tests build success")
