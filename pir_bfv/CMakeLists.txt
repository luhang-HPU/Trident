option(PIR_USE_HARDWARE "USE HARDWARE TO RUN PIR of BFV" OFF)

if(PIR_USE_HARDWARE)
    message(STATUS "pir_bfv hardware version")
    add_definitions(-DPIR_USE_HARDWARE)
else()
    message(STATUS "pir_bfv software version")
endif()

file(GLOB PIR_BFV_SOURCE_FILES "*.cpp" "*.c")
add_executable(pir_bfv ${PIR_BFV_SOURCE_FILES})
target_link_libraries(pir_bfv PUBLIC poseidon_shared)
message(STATUS "PIR_BFV build success")

option(PIR_BFV_BUILD_TESTS "Build tests for PIR of BFV" OFF)
if(PIR_BFV_BUILD_TESTS)
    add_executable(pir_bfv_tests)
    add_subdirectory(tests)
    target_include_directories(pir_bfv_tests PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}>)
    target_link_libraries(pir_bfv_tests poseidon_shared GTest::gtest)
endif()
message(STATUS "PIR_BFV tests build success")
